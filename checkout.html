/* ========================================================= AMIANI'S FLY WHOLESALERS CHECKOUT - USD pricing - Prices are per dozen - Supports product size + color variants - Saves variants into order items ========================================================= */ (function () { "use strict"; const $ = (selector, root = document) => root.querySelector(selector); const checkoutItems = $("#checkoutItems"); const checkoutTotal = $("#checkoutTotal"); const checkoutForm = $("#checkoutForm"); const orderMsg = $("#orderMsg"); /* --------------------------------------------------------- Helpers --------------------------------------------------------- */ function money(value) { return new Intl.NumberFormat("en-US", { style: "currency", currency: "USD" }).format(Number(value || 0)); } function esc(value) { return String(value ?? "") .replace(/&/g, "&") .replace(//g, ">") .replace(/"/g, """) .replace(/'/g, "'"); } function showMessage(message, type = "notice") { if (!orderMsg) return; orderMsg.innerHTML = `
${esc(message)}
`; } function getCart() { try { const saved = localStorage.getItem("amani_cart"); if (!saved) { return []; } const parsed = JSON.parse(saved); return Array.isArray(parsed) ? parsed : []; } catch (error) { console.error("Could not read cart:", error); return []; } } function clearCart() { localStorage.removeItem("amani_cart"); if (window.AMIANI && typeof window.AMIANI.updateCartCount === "function") { window.AMIANI.updateCartCount(); } } /* --------------------------------------------------------- Render checkout items --------------------------------------------------------- */ function renderCheckout() { const cart = getCart(); if (!checkoutItems || !checkoutTotal) { return; } if (!cart.length) { checkoutItems.innerHTML = `
Your cart is empty.
Continue Shopping
`; checkoutTotal.textContent = "$0.00"; if (checkoutForm) { const submitButton = checkoutForm.querySelector( 'button[type="submit"]' ); if (submitButton) { submitButton.disabled = true; submitButton.textContent = "Cart is Empty"; } } return; } let total = 0; checkoutItems.innerHTML = cart.map((item) => { const price = Number(item.price || 0); const qty = Math.max(1, Number(item.qty || 1)); const lineTotal = price * qty; total += lineTotal; const image = item.image_url ? ` ` : `
No Image
`; const variantParts = []; if (item.size) { variantParts.push( `Size: ${esc(item.size)}` ); } if (item.color) { variantParts.push( `Color: ${esc(item.color)}` ); } const variants = variantParts.length ? `
${variantParts.join("")}
` : ""; return `
${image}

${esc(item.name)}

${variants}
${money(price)} / dozen × ${qty} dozen
${money(lineTotal)}
`; }).join(""); checkoutTotal.textContent = money(total); if (checkoutForm) { const submitButton = checkoutForm.querySelector( 'button[type="submit"]' ); if (submitButton) { submitButton.disabled = false; submitButton.textContent = "Place Order"; } } } /* --------------------------------------------------------- Calculate order total --------------------------------------------------------- */ function calculateTotal(cart) { return cart.reduce((total, item) => { const price = Number(item.price || 0); const qty = Math.max(1, Number(item.qty || 1)); return total + (price * qty); }, 0); } /* --------------------------------------------------------- Submit order --------------------------------------------------------- */ async function submitOrder(event) { event.preventDefault(); if (!checkoutForm) { return; } const cart = getCart(); if (!cart.length) { showMessage( "Your cart is empty. Please add products before checking out.", "notice" ); return; } /* ----------------------------------------------------- Supabase check ----------------------------------------------------- */ const supabaseClient = window.AMANI_SUPABASE || (window.AMIANI && window.AMIANI.sb); if (!supabaseClient) { showMessage( "The store connection is not available. Please refresh the page and try again.", "notice" ); console.error( "Supabase client is not available." ); return; } /* ----------------------------------------------------- Form values ----------------------------------------------------- */ const formData = new FormData(checkoutForm); const customerName = String(formData.get("customer_name") || "").trim(); const phone = String(formData.get("phone") || "").trim(); const email = String(formData.get("email") || "").trim(); const address = String(formData.get("address") || "").trim(); const notes = String(formData.get("notes") || "").trim(); if (!customerName) { showMessage( "Please enter your full name.", "notice" ); return; } if (!phone) { showMessage( "Please enter your phone number.", "notice" ); return; } if (!address) { showMessage( "Please enter your delivery address.", "notice" ); return; } /* ----------------------------------------------------- Calculate total ----------------------------------------------------- */ const total = calculateTotal(cart); if (!Number.isFinite(total) || total <= 0) { showMessage( "Your order total is invalid. Please check your cart.", "notice" ); return; } /* ----------------------------------------------------- Prepare order items IMPORTANT: size + color are saved here. ----------------------------------------------------- */ const orderItems = cart.map((item) => ({ id: item.id || null, name: String(item.name || ""), price: Number(item.price || 0), qty: Math.max( 1, Number(item.qty || 1) ), image_url: item.image_url || "", category_id: item.category_id || item.categoryId || null, size: item.size ? String(item.size) : "", color: item.color ? String(item.color) : "" })); /* ----------------------------------------------------- Disable button ----------------------------------------------------- */ const submitButton = checkoutForm.querySelector( 'button[type="submit"]' ); if (submitButton) { submitButton.disabled = true; submitButton.textContent = "Placing Order..."; } showMessage( "Submitting your order...", "notice" ); try { /* ------------------------------------------------- Insert order ------------------------------------------------- */ const { data, error } = await supabaseClient .from("orders") .insert([ { customer_name: customerName, phone: phone, email: email || null, address: address, notes: notes || null, items: orderItems, total: total, status: "pending" } ]) .select() .single(); if (error) { console.error( "Order insert error:", error ); throw error; } console.log( "Order created successfully:", data ); /* ------------------------------------------------- Clear cart ------------------------------------------------- */ clearCart(); /* ------------------------------------------------- Success message ------------------------------------------------- */ checkoutItems.innerHTML = `
✓

Order Received

Thank you for ordering from Amiani's Fly Wholesalers.

${ data && data.id ? `

Order reference: ${esc(data.id)}

` : "" }
`; checkoutTotal.textContent = money(total); showMessage( "Your order has been placed successfully. We will contact you to confirm the order and delivery details.", "notice" ); checkoutForm.reset(); if (submitButton) { submitButton.disabled = true; submitButton.textContent = "Order Placed"; } } catch (error) { console.error( "Checkout error:", error ); showMessage( error.message || "We could not place your order. Please try again.", "notice" ); if (submitButton) { submitButton.disabled = false; submitButton.textContent = "Place Order"; } } } /* --------------------------------------------------------- Listen for cart changes --------------------------------------------------------- */ window.addEventListener( "storage", function (event) { if (event.key === "amani_cart") { renderCheckout(); } } ); /* --------------------------------------------------------- Initialize --------------------------------------------------------- */ function initCheckout() { if (!checkoutItems) { return; } renderCheckout(); if (checkoutForm) { checkoutForm.addEventListener( "submit", submitOrder ); } } /* app.js is loaded with defer before checkout.js, so DOMContentLoaded is safe here. */ if ( document.readyState === "loading" ) { document.addEventListener( "DOMContentLoaded", initCheckout ); } else { initCheckout(); } })();
